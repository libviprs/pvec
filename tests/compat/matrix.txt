# The compatibility matrix (issue P7.6): old reader x new file and new reader x old file.
#
# Each case declares the outcome the compatibility rules of docs/COMPATIBILITY.md give. The outcome is
# written here by hand and never computed by the harness, so a change in behaviour turns a case red.
#
# reader      s_old = geometry 1..13 at 1.0, no component stream, reader version 1.0.0
#             s_new = s_old plus every component stream plus capability 99 up to 3.1, reader version 1.1.0
#             v1    = this build, every capability of the v1 registry
# base        the file the case starts from: "mixed" (geometry and the five component streams)
# caps        capabilities added to, or replaced in, the base manifest: id:requirement:major.minor
# min_reader  the minimum reader version the manifest declares (default 1.0.0)
# garbage     a block of random bytes with a valid length: kind:capability:length
# outcome     open | open_degraded | refuse
# rule        for refuse, the NEGOTIATE rule of docs/CONTAINER.md
# named       for refuse, the capability id the refusal names
# skipped     for open_degraded, the skipped capability ids
# reasons     for open_degraded, the skip reason of each skipped capability, in the same order
# blocks      for open_degraded, the skipped blocks as kind:reason, in directory order (default none)
# gaps        capabilities read at a newer minor than the reader implements: id:file-version
# geometry    same = the geometry decoded is identical to the base file's

case old-reader-new-file-unknown-optional-block
reader = s_old
base = mixed
caps = 99:optional:3.1
garbage = 9:99:64
outcome = open_degraded
skipped = 14,15,16,18,20,99
reasons = not_implemented,not_implemented,not_implemented,not_implemented,not_implemented,not_implemented
blocks = 9:unknown_block_kind
geometry = same

case old-reader-new-file-unknown-required-capability
reader = s_old
base = mixed
caps = 99:required:2.1
outcome = refuse
rule = NEGOTIATE-01
named = 99

case old-reader-file-with-component-streams

base = mixed
outcome = open_degraded
skipped = 14,15,16,18,20
reasons = not_implemented,not_implemented,not_implemented,not_implemented,not_implemented
geometry = same

case old-reader-new-file-two-unknown-required-names-the-first
reader = s_old
base = mixed
caps = 98:required:1.0,99:required:1.0
outcome = refuse
rule = NEGOTIATE-01
named = 98
